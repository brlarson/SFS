\begin{restatable}{metadefinition}{dfbltime}~\blTitle{df-bl.time}~ Define domain of time $\mathbf{T}$
from $\msc{t} = 0$ to $\mathsf{\mc{now}}$.
\begin{align}
\vdash \mathbf{T} = \{ \msc{x}~\vert~( \msc{x} \in \mathbb{R} \wedge 0 \le \msc{x}
    \wedge \msc{x} \le \mathsf{\mc{now}} )
    \}\label{eq:df-bl.time}\tag{df-bl.time}
\end{align}
\end{restatable}

\begin{restatable}{lemma}{blrt}~\blTitle{bl.rt}~ $\mathbf{T}$ is $\mathbb{R}$ (real)
\begin{align}
\vdash \mathbf{T} \subseteq \mathbb{R}\label{eq:bl.rt}\tag{bl.rt}
\end{align}
\end{restatable}

\begin{proof}
\begin{align}
  1 &&  & \vdash \mathbf{T} = \{ \msc{x}~\vert~( \msc{x} \in \mathbb{R} \wedge 0
      \le \msc{x} \wedge \msc{x} \le \mathsf{\mc{now}} )
      \}&(\mbox{\ref{eq:df-bl.time}})\notag
  \\ 2 &&  & \vdash ( \msc{x} \in \mathbf{T} \leftrightarrow \msc{x} \in \{
      \msc{x}~\vert~( \msc{x} \in \mathbb{R} \wedge 0 \le
      \msc{x} \wedge \msc{x} \le \mathsf{\mc{now}} ) \}
      )&(\mbox{\ref{eq:eleq2i}},1)\notag
  \\ 3 &&  & \vdash ( \msc{x} \in \{ \msc{x}~\vert~( \msc{x} \in \mathbb{R} \wedge
      0 \le \msc{x} \wedge \msc{x} \le \mathsf{\mc{now}} )
      \} \leftrightarrow ( \msc{x} \in \mathbb{R} \wedge 0
      \le \msc{x} \wedge \msc{x} \le \mathsf{\mc{now}} )
      )&(\mbox{\ref{eq:abid}})\notag
  \\ 4 &&  & \vdash ( \msc{x} \in \mathbf{T} \leftrightarrow ( \msc{x} \in
      \mathbb{R} \wedge 0 \le \msc{x} \wedge \msc{x} \le
      \mathsf{\mc{now}} )
      )&(\mbox{\ref{eq:bitri}},2,3)\notag
  \\ 5 &&  & \vdash ( \msc{x} \in \mathbf{T} \rightarrow \msc{x} \in \mathbb{R}
      )&(\mbox{\ref{eq:simp1bi}},4)\notag
  \\ 6 &&  & \vdash \mathbf{T} \subseteq
      \mathbb{R}&(\mbox{\ref{eq:ssriv}},5)\notag
\end{align}
\end{proof}

\begin{restatable}{lemma}{bletial0}~\blTitle{bl.etial0}~ Every time is at least 0.

      \exists \msc{x} ( \msc{x} = \tau_1

  \\ 2 &&  & \vdash \mathbf{T} = \{ \msc{x}~\vert~( \msc{x} \in \mathbb{R} \wedge 0
      \le \msc{x} \wedge \msc{x} \le \mathsf{\mc{now}} )
      \}&(\mbox{\ref{eq:df-bl.time}})\notag
  \\ 3 &&  & \vdash ( \tau_1 \in \mathbf{T} \leftrightarrow \tau_1 \in \{
      \msc{x}~\vert~( \msc{x} \in \mathbb{R} \wedge 0 \le
      \msc{x} \wedge \msc{x} \le \mathsf{\mc{now}} ) \}
      )&(\mbox{\ref{eq:eleq2i}},2)\notag
  \\ 4 &&  & \vdash ( \underaccent{\dot}{[} \tau_1 / \msc{x}
      \underaccent{\dot}{]} ( \msc{x} \in \mathbb{R} \wedge
      0 \le \msc{x} \wedge \msc{x} \le \mathsf{\mc{now}} )
      \leftrightarrow \exists \msc{x} ( \msc{x} = \tau_1
      \wedge ( \msc{x} \in \mathbb{R} \wedge 0 \le \msc{x}
      \wedge \msc{x} \le \mathsf{\mc{now}} ) )
      )&(\mbox{\ref{eq:sbc5}})\notag
  \\ 5 &&  & \vdash ( \tau_1 \in \mathbf{T} \leftrightarrow
      \underaccent{\dot}{[} \tau_1 / \msc{x}
      \underaccent{\dot}{]} ( \msc{x} \in \mathbb{R} \wedge
      0 \le \msc{x} \wedge \msc{x} \le \mathsf{\mc{now}} )
      )&(\mbox{\ref{eq:3bitr4i}},1,3,4)\notag
  \\ 6 &&  & \vdash ( \underaccent{\dot}{[} \tau_1 / \msc{x}
      \underaccent{\dot}{]} ( \msc{x} \in \mathbb{R} \wedge
      0 \le \msc{x} \wedge \msc{x} \le \mathsf{\mc{now}} )
      \leftrightarrow \notag \\ & & & \qquad
       ( \underaccent{\dot}{[} \tau_1 /
      \msc{x} \underaccent{\dot}{]} \msc{x} \in \mathbb{R}
      \wedge \underaccent{\dot}{[} \tau_1 / \msc{x}
      \underaccent{\dot}{]} 0 \le \msc{x} \wedge
      \underaccent{\dot}{[} \tau_1 / \msc{x}
      \underaccent{\dot}{]} \msc{x} \le \mathsf{\mc{now}} )
      )&(\mbox{\ref{eq:sbc3an}})\notag
  \\ 7 &&  & \vdash ( \tau_1 \in \mathbf{T} \leftrightarrow (
      \underaccent{\dot}{[} \tau_1 / \msc{x}
      \underaccent{\dot}{]} \msc{x} \in \mathbb{R} \wedge
      \underaccent{\dot}{[} \tau_1 / \msc{x}
      \underaccent{\dot}{]} 0 \le \msc{x} \wedge
      \underaccent{\dot}{[} \tau_1 / \msc{x}
      \underaccent{\dot}{]} \msc{x} \le \mathsf{\mc{now}} )
      )&(\mbox{\ref{eq:bitri}},5,6)\notag
  \\ 8 &&  & \vdash ( \tau_1 \in \mathbf{T} \rightarrow \underaccent{\dot}{[}
      \tau_1 / \msc{x} \underaccent{\dot}{]} 0 \le \msc{x}
      )&(\mbox{\ref{eq:simp2bi}},7)\notag
  \\ 9 &&  & \vdash ( \tau_1 \in \mathbf{T} \rightarrow ( \underaccent{\dot}{[}
      \tau_1 / \msc{x} \underaccent{\dot}{]} 0 \le \msc{x}
      \leftrightarrow 0 \le \underline{[} \tau_1 / \msc{x}
      \underline{]} \msc{x} )

  \\ 10 &&  & \vdash ( \tau_1 \in \mathbf{T} \rightarrow 0 \le \underline{[}

  \\ 11 &&  & \vdash ( \tau_1 \in \mathbf{T} \rightarrow 0 = 0
      )&(\mbox{\ref{eq:eqidd}})\notag
  \\ 12 &&  & \vdash ( \tau_1 \in \mathbf{T} \rightarrow \underline{[} \tau_1 /

  \\ 13 &&  & \vdash ( \tau_1 \in \mathbf{T} \rightarrow ( 0 \le \underline{[}

  \\ 14 &&  & \vdash ( \tau_1 \in \mathbf{T} \rightarrow 0 \le \tau_1
