\pn The \texttt{\^{}.} operator shifts an instant by a whole multiple of a step size, which can carry it outside $\mathbf{T}$ even when the starting instant is inside it --- exactly why \texttt{SFS.mm}'s own \texttt{df-bl.ts}/\texttt{bl.tsi} carry a $(\tau_0+D\times n)\in\mathbf{T}$ side condition. \texttt{shiftEval}/\texttt{shiftEvalC} take that landing-in-$\mathbf{T}$ proof as an explicit argument, rather than defaulting to an unshifted or junk value when it fails (tried and rejected: no single default keeps negation consistent on both sides).

\begin{restatable}{leandefinition}{ShiftEvalShiftEvalC}~\leanTitle{shiftEval / shiftEvalC}~ \texttt{SFS.mm} \texttt{wts}/\texttt{clats}: evaluate $\varphi$/\texttt{A} shifted by \texttt{n} whole multiples of step size \texttt{D} from \texttt{t0}, given that the shift lands back in $\mathbf{T}$.

\texttt{\textcolor{leancolor}{def} \textcolor{leanyellow}{shiftEval} ($\varphi$ : TProp) (t0 : Time) (D : $\mathbb{R}$) (n : $\mathbb{Z}$) (h : t0.val + D * n $\in$ TIME) : Prop :=}

\texttt{~~$\varphi$ $\langle$t0.val + D * n, h$\rangle$}

\texttt{\textcolor{leancolor}{def} \textcolor{leanyellow}{shiftEvalC} \{$\alpha$\} (A : TVal $\alpha$) (t0 : Time) (D : $\mathbb{R}$) (n : $\mathbb{Z}$) (h : t0.val + D * n $\in$ TIME) : $\alpha$ :=}

\texttt{~~A $\langle$t0.val + D * n, h$\rangle$}
\label{shiftEval}
\label{shiftEvalC}
\end{restatable}

\begin{restatable}{metadefinition}{dfblts}~\blTitle{df-bl.ts}~ Definition of time-shift predicate.
\begin{align}
\vdash ( ( \mc{A} \in \mathbb{Z} \wedge ( \mc{D} \in \mathbb{R} \wedge \tau \in
    \mathbf{T} \wedge ( \tau + ( \mc{D} \times \mc{A} ) ) \in \mathbf{T} ) )

    \times \mc{A} ) ) \rsem ) )\label{eq:df-bl.ts}\tag{df-bl.ts}
\end{align}
\end{restatable}
