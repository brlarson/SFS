  \\ 4 &&  & \vdash ( \msc{t} \in \mathbf{T} \rightarrow ( ( ( \mw{\varphi}
     \wedge ( \mw{\psi} \wedge \mw{\chi} ) ) @ \msc{t} )
     \leftrightarrow ( ( \mw{\varphi} @ \msc{t} ) \wedge (
     ( \mw{\psi} @ \msc{t} ) \wedge ( \mw{\chi} @ \msc{t}
     ) ) ) ) )&(\mbox{\ref{eq:bitrd}},1,3)\notag
\end{align}
\end{proof}
