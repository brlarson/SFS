\begin{lemma}\blTitle{bl.an2wl} two-term list is conjunction
\begin{align}
\vdash ( \bigwedge( {\color{formulacolor}\varphi} {\color{formulacolor}\psi} )
    \leftrightarrow ( {\color{formulacolor}\varphi} \wedge
    {\color{formulacolor}\psi} ) )\label{bl.an2wl}
\end{align}
\end{lemma}

\begin{proof}
\begin{align}
  1 && & \vdash ( \bigwedge( {\color{formulacolor}\varphi}
      {\color{formulacolor}\psi} ) \leftrightarrow (
      \bigwedge( {\color{formulacolor}\varphi} ) \wedge
      \bigwedge( {\color{formulacolor}\psi} ) )
      )&(\mmref{df-lan2wl})\notag
  \\ 2 && & \vdash ( \bigwedge( {\color{formulacolor}\varphi} ) \leftrightarrow
      {\color{formulacolor}\varphi}
      )&(\mmref{df-lan1})\notag
  \\ 3 && & \vdash ( \bigwedge( {\color{formulacolor}\psi} ) \leftrightarrow
      {\color{formulacolor}\psi} )&(\mmref{df-lan1})\notag
  \\ 4 && & \vdash ( ( \bigwedge( {\color{formulacolor}\varphi} ) \wedge
      \bigwedge( {\color{formulacolor}\psi} ) )
      \leftrightarrow ( {\color{formulacolor}\varphi}
      \wedge {\color{formulacolor}\psi} )
      )&2,3,(\mmref{anbi12i})\notag
  \\ 5 && & \vdash ( \bigwedge( {\color{formulacolor}\varphi}
      {\color{formulacolor}\psi} ) \leftrightarrow (
      {\color{formulacolor}\varphi} \wedge
      {\color{formulacolor}\psi} )
      )&1,4,(\mmref{bitri})\notag
\end{align}
\end{proof}
