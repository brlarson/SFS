\begin{restatable}{metadefinition}{dfstruct}~\otherTitle{df-struct}~ Define a structure with components in
$M \ldots N$.    ~
\begin{align}
\vdash \,\mathrm{Struct} = \{ \langle \msc{f} , \msc{x} \rangle~\vert~( \msc{x} \in
    ( \le \cap ( \mathbb{N} \times \mathbb{N} ) ) \wedge \,\mathrm{Fun}~ (
    \msc{f} \setminus \{ \varnothing~ \} ) \wedge \,\mathrm{dom}~ \msc{f}
    \subseteq ( \ldots ` \msc{x} ) ) \}\label{eq:df-struct}\tag{df-struct}
\end{align}
\end{restatable}

\begin{restatable}{metadefinition}{dfslot}~\otherTitle{df-slot}~ Define the slot extractor for extensible
structures.  The class
       $\,\mathrm{Slot}~ \mc{A}$ is a function whose argument can be any set,
although it is
       meaningful only if that set is a member of an extensible structure
.  ~
\begin{align}
\vdash \,\mathrm{Slot}~ \mc{A} = ( \msc{x} \in \,\mathrm{V} \mapsto ( \msc{x} `
    \mc{A} ) )\label{eq:df-slot}\tag{df-slot}
\end{align}
\end{restatable}

\begin{restatable}{metadefinition}{dfsets}~\otherTitle{df-sets}~ Set a component of an extensible
structure.  This function is useful for
       taking an existing structure and ``overriding" one of its components.
        ~
\begin{align}
\vdash \,\mathrm{sSet} = ( \msc{s} \in \,\mathrm{V} , \msc{e} \in \,\mathrm{V}
    \mapsto ( ( \msc{s} \restriction ( \,\mathrm{V} \setminus \,\mathrm{dom}~ \{
    \msc{e} \} ) ) \cup \{ \msc{e} \} ) )\label{eq:df-sets}\tag{df-sets}
\end{align}
\end{restatable}

\begin{restatable}{lemma}{isstruct}~\otherTitle{isstruct}~ The property of being a structure with
components in $M \ldots N$.
     ~
\begin{align}
\vdash ( F \,\mathrm{Struct} \langle M , N \rangle \leftrightarrow \notag \\
( ( M \in
    \mathbb{N} \wedge N \in \mathbb{N} \wedge M \le N ) \wedge \,\mathrm{Fun}~ (
    F \setminus \{ \varnothing~ \} ) \wedge \,\mathrm{dom}~ F \subseteq ( M
    \ldots N ) ) )\label{eq:isstruct}\tag{isstruct}
\end{align}
\end{restatable}
