\begin{restatable}{axiom}{axbltaut}~\blTitle{ax-bl.taut}~ Tautologies hold in $\mathfrak{M}$.
   ax-bl.taut syas that if $\mw{\varphi}$ is tautology, it also holds in
models, $\,\mathrm{mathhfrakM}$.
\begin{align}
\vdash ( \mw{\varphi} \leftrightarrow ( \mathfrak{M} \models \mw{\varphi} )
    )\label{eq:ax-bl.taut}\tag{ax-bl.taut}
\end{align}
\end{restatable}

\begin{restatable}{axiom}{axblmodels}~\blTitle{ax-bl.models}~ Defing $\mathfrak{M}$ models predicate
for any $\msc{t} \in \mathbf{T}$
\begin{align}
\vdash \msc{t} \in \mathbf{T}\quad\Rightarrow \notag \\ 
	\quad \vdash ( ( \mathfrak{M}
    \models \mw{\varphi} ) \rightarrow ( \mathfrak{M} , \msc{t} \models
    \mw{\varphi} ) )\label{eq:ax-bl.models}\tag{ax-bl.models}
\end{align}
\end{restatable}

\begin{restatable}{axiom}{axblmodelsc}~\blTitle{ax-bl.modelsc}~ Defing $\mathfrak{M}$ models values
over any $\msc{t} \in \mathbf{T}$
\begin{align}
\vdash \msc{t} \in \mathbf{T}\quad\Rightarrow \notag \\ 
	\quad \vdash ( ( \mathfrak{M}
    \models \mc{A} ) = \mc{B} \rightarrow ( \mathfrak{M} , \msc{t} \models
    \mc{A} ) = \mc{B} )\label{eq:ax-bl.modelsc}\tag{ax-bl.modelsc}
\end{align}
\end{restatable}

\begin{restatable}{axiom}{axblbi}~\blTitle{ax-bl.bi}~ Axiom of temporal equivalence.

    \msc{t_1} = \msc{t_2} ) \rightarrow ( ( \mw{\varphi} \leftrightarrow

\end{align}
\end{restatable}

\begin{restatable}{lemma}{blfrextwo}~\blTitle{bl.dfrex2}~ Define existential quantification in
$\mathfrak{M}$.
\begin{align}
\vdash ( ( \mathfrak{M} \models \exists \msc{x} \in \mc{A} \mw{\varphi} )
    \leftrightarrow ( \mathfrak{M} \models \lnot \forall \msc{x} \in \mc{A}
    \lnot \mw{\varphi} ) )\label{eq:bl.dfrex2}\tag{bl.dfrex2}
\end{align}
\end{restatable}

\begin{proof}
\begin{align}
  1 &&  & \vdash ( \exists \msc{x} \in \mc{A} \mw{\varphi} \leftrightarrow
     ~\lnot \forall \msc{x} \in \mc{A} \lnot \mw{\varphi}
     ~)&(\mbox{\ref{eq:dfrex2}})\notag
  \\ 2 &&  & \vdash ( ( \exists \msc{x} \in \mc{A} \mw{\varphi} \leftrightarrow
     ~\lnot \forall \msc{x} \in \mc{A} \lnot \mw{\varphi} )
     ~\leftrightarrow ( \mathfrak{M} \models ( \exists
     ~\msc{x} \in \mc{A} \mw{\varphi} \leftrightarrow \lnot
     ~\forall \msc{x} \in \mc{A} \lnot \mw{\varphi} ) )
     ~)&(\mbox{\ref{eq:ax-bl.taut}})\notag
  \\ 3 &&  & \vdash ( \mathfrak{M} \models ( \exists \msc{x} \in \mc{A}
     ~\mw{\varphi} \leftrightarrow \lnot \forall \msc{x}
     ~\in \mc{A} \lnot \mw{\varphi} )
     ~)&(\mbox{\ref{eq:mpbi}},1,2)\notag
  \\ 4 &&  & \vdash ( ( \mathfrak{M} \models ( \exists \msc{x} \in \mc{A}
     ~\mw{\varphi} \leftrightarrow \lnot \forall \msc{x}
     ~\in \mc{A} \lnot \mw{\varphi} ) ) \leftrightarrow ( (
     ~\mathfrak{M} \models \exists \msc{x} \in \mc{A}
     ~\mw{\varphi} ) \leftrightarrow ( \mathfrak{M} \models
     ~\lnot \forall \msc{x} \in \mc{A} \lnot \mw{\varphi} )
     ~) )&(\mbox{\ref{eq:df-bl.bi}})\notag
  \\ 5 &&  & \vdash ( ( \mathfrak{M} \models \exists \msc{x} \in \mc{A}
     ~\mw{\varphi} ) \leftrightarrow ( \mathfrak{M} \models
     ~\lnot \forall \msc{x} \in \mc{A} \lnot \mw{\varphi} )
     ~)&(\mbox{\ref{eq:mpbi}},3,4)\notag
\end{align}
\end{proof}

\begin{restatable}{lemma}{blex}~\blTitle{bl.ex}~ Exportation of existential quantification
from $\mathfrak{M}$
\begin{align}
\vdash ( ( \mathfrak{M} \models \exists \msc{x} \in \mc{A} \mw{\varphi} )
    \leftrightarrow \exists \msc{x} \in \mc{A} ( \mathfrak{M} \models
    \mw{\varphi} ) )\label{eq:bl.ex}\tag{bl.ex}
\end{align}
\end{restatable}

\begin{proof}
\begin{align}
  1 &&  & \vdash ( ( \mathfrak{M} \models \exists \msc{x} \in \mc{A}
     ~\mw{\varphi} ) \leftrightarrow ( \mathfrak{M} \models
     ~\lnot \forall \msc{x} \in \mc{A} \lnot \mw{\varphi} )
     ~)&(\mbox{\ref{eq:bl.dfrex2}})\notag
  \\ 2 &&  & \vdash ( ( \mathfrak{M} \models \lnot \forall \msc{x} \in \mc{A}
     ~\lnot \mw{\varphi} ) \leftrightarrow \lnot (
     ~\mathfrak{M} \models \forall \msc{x} \in \mc{A} \lnot
     ~\mw{\varphi} ) )&(\mbox{\ref{eq:df-bl.not}})\notag
  \\ 3 &&  & \vdash ( ( \mathfrak{M} \models \exists \msc{x} \in \mc{A}
     ~\mw{\varphi} ) \leftrightarrow \lnot ( \mathfrak{M}
     ~\models \forall \msc{x} \in \mc{A} \lnot \mw{\varphi}
     ~) )&(\mbox{\ref{eq:bitri}},1,2)\notag
  \\ 4 &&  & \vdash ( ( \mathfrak{M} \models \forall \msc{x} \in \mc{A} \lnot
     ~\mw{\varphi} ) \leftrightarrow \forall \msc{x} \in
     ~\mc{A} ( \mathfrak{M} \models \lnot \mw{\varphi} )
     ~)&(\mbox{\ref{eq:df-bl.al}})\notag
  \\ 5 &&  & \vdash ( \lnot ( \mathfrak{M} \models \forall \msc{x} \in \mc{A}
     ~\lnot \mw{\varphi} ) \leftrightarrow \lnot \forall
     ~\msc{x} \in \mc{A} ( \mathfrak{M} \models \lnot
     ~\mw{\varphi} ) )&(\mbox{\ref{eq:notbii}},4)\notag
  \\ 6 &&  & \vdash ( ( \mathfrak{M} \models \exists \msc{x} \in \mc{A}
     ~\mw{\varphi} ) \leftrightarrow \lnot \forall \msc{x}
     ~\in \mc{A} ( \mathfrak{M} \models \lnot \mw{\varphi}
     ~) )&(\mbox{\ref{eq:bitri}},3,5)\notag
  \\ 7 &&  & \vdash ( ( \mathfrak{M} \models \lnot \mw{\varphi} )
     ~\leftrightarrow \lnot ( \mathfrak{M} \models
     ~\mw{\varphi} ) )&(\mbox{\ref{eq:df-bl.not}})\notag
  \\ 8 &&  & \vdash ( \forall \msc{x} \in \mc{A} ( \mathfrak{M} \models \lnot
     ~\mw{\varphi} ) \leftrightarrow \forall \msc{x} \in
     ~\mc{A} \lnot ( \mathfrak{M} \models \mw{\varphi} )
     ~)&(\mbox{\ref{eq:ralbii}},7)\notag
  \\ 9 &&  & \vdash ( \lnot \forall \msc{x} \in \mc{A} ( \mathfrak{M} \models
     ~\lnot \mw{\varphi} ) \leftrightarrow \lnot \forall
     ~\msc{x} \in \mc{A} \lnot ( \mathfrak{M} \models
     ~\mw{\varphi} ) )&(\mbox{\ref{eq:notbii}},8)\notag
  \\ 10 &&  & \vdash ( ( \mathfrak{M} \models \exists \msc{x} \in \mc{A}
     ~\mw{\varphi} ) \leftrightarrow \lnot \forall \msc{x}
     ~\in \mc{A} \lnot ( \mathfrak{M} \models \mw{\varphi}
     ~) )&(\mbox{\ref{eq:bitri}},6,9)\notag
  \\ 11 &&  & \vdash ( \exists \msc{x} \in \mc{A} ( \mathfrak{M} \models
     ~\mw{\varphi} ) \leftrightarrow \lnot \forall \msc{x}
     ~\in \mc{A} \lnot ( \mathfrak{M} \models \mw{\varphi}
     ~) )&(\mbox{\ref{eq:dfrex2}})\notag
  \\ 12 &&  & \vdash ( ( \mathfrak{M} \models \exists \msc{x} \in \mc{A}
     ~\mw{\varphi} ) \leftrightarrow \exists \msc{x} \in
     ~\mc{A} ( \mathfrak{M} \models \mw{\varphi} )
     ~)&(\mbox{\ref{eq:bitr4i}},10,11)\notag
\end{align}
\end{proof}
