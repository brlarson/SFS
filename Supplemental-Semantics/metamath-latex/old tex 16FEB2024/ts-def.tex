\begin{restatable}{axiom}{axblts}~\blTitle{ax-bl.ts}~ Time shift predicate.
\begin{align}
\vdash ( ( \mc{A} \in \mathbb{Z} \wedge ( \mc{D} \in \mathbb{R} \wedge \msc{t}
    \in \mathbf{T} \wedge ( \msc{t} + ( \mc{D} \cdot \mc{A} ) ) \in \mathbf{T}
    ) ) \rightarrow ( ( \mathfrak{M} , \msc{t} \models ( \mw{\varphi}
    ~\string^~ \mc{A} ) ) \leftrightarrow ( \mathfrak{M} , ( \msc{t} + ( \mc{D}
    \cdot \mc{A} ) ) \models \mw{\varphi} ) )
    )\label{eq:ax-bl.ts}\tag{ax-bl.ts}
\end{align}
\end{restatable}
