\begin{restatable}{lemma}{blalldd}~\blTitle{bl.alldd}~ Universal quantification over closed time
interval [t1 .. t2].
\begin{align}
\vdash ( \msc{t_1} \in \mathbf{T} \wedge \msc{t_2} \in \mathbf{T}
    )\quad\Rightarrow \notag \\ 
	\quad \vdash ( \forall \msc{x} \in \mathbf{T} ( \msc{x}
    \in ( \msc{t_1} [,] \msc{t_2} ) \wedge \mw{\varphi} ) \leftrightarrow
    \forall \msc{x} \in \mathbf{T} ( \msc{t_1} \le \msc{x} \wedge \msc{x} \le
    \msc{t_2} \wedge \mw{\varphi} ) )\label{eq:bl.alldd}\tag{bl.alldd}
\end{align}
\end{restatable}

\begin{proof}
\begin{align}
  1 &&  & \vdash ( \msc{t_1} \in \mathbf{T} \wedge \msc{t_2} \in \mathbf{T}
     ~)&\text{Hyp~alldd.1}\notag%bl.alldd.1
  \\ 2 &&  & \vdash ( ( \msc{t_1} \in \mathbf{T} \wedge \msc{t_2} \in
     ~\mathbf{T} ) \rightarrow \msc{t_1} \in \mathbf{T}
     ~)&(\mbox{\ref{eq:simpl}})\notag
  \\ 3 &&  & \vdash \msc{t_1} \in \mathbf{T}&(\mbox{\ref{eq:ax-mp}},1,2)\notag
  \\ 4 &&  & \vdash \mathbf{T} \subseteq
     ~\mathbb{R}^{*}&(\mbox{\ref{eq:bl.ert}})\notag

     ~\mathbb{R}^{*} )&(\mbox{\ref{eq:sseli}},4)\notag
  \\ 6 &&  & \vdash \msc{t_1} \in
     ~\mathbb{R}^{*}&(\mbox{\ref{eq:ax-mp}},3,5)\notag
  \\ 7 &&  & \vdash ( \msc{t_1} \in \mathbf{T} \wedge \msc{t_2} \in \mathbf{T}
     ~)&\text{Hyp~alldd.1}\notag%bl.alldd.1
  \\ 8 &&  & \vdash ( ( \msc{t_1} \in \mathbf{T} \wedge \msc{t_2} \in
     ~\mathbf{T} ) \rightarrow \msc{t_2} \in \mathbf{T}
     ~)&(\mbox{\ref{eq:simpr}})\notag
  \\ 9 &&  & \vdash \msc{t_2} \in \mathbf{T}&(\mbox{\ref{eq:ax-mp}},7,8)\notag
  \\ 10 &&  & \vdash \mathbf{T} \subseteq
     ~\mathbb{R}^{*}&(\mbox{\ref{eq:bl.ert}})\notag
  \\ 11 &&  & \vdash ( \msc{t_2} \in \mathbf{T} \rightarrow \msc{t_2} \in
     ~\mathbb{R}^{*} )&(\mbox{\ref{eq:sseli}},10)\notag
  \\ 12 &&  & \vdash \msc{t_2} \in
     ~\mathbb{R}^{*}&(\mbox{\ref{eq:ax-mp}},9,11)\notag
  \\ 13 &&  & \vdash ( \msc{t_1} \in \mathbb{R}^{*} \wedge \msc{t_2} \in
     ~\mathbb{R}^{*} )&(\mbox{\ref{eq:pm3.2i}},6,12)\notag
  \\ 14 &&  & \vdash ( ( \msc{t_1} \in \mathbb{R}^{*} \wedge \msc{t_2} \in
     ~\mathbb{R}^{*} ) \rightarrow ( \msc{x} \in (
     ~\msc{t_1} [,] \msc{t_2} ) \leftrightarrow ( \msc{x}
     ~\in \mathbb{R}^{*} \wedge \msc{t_1} \le \msc{x}
     ~\wedge \msc{x} \le \msc{t_2} ) )

  \\ 15 &&  & \vdash ( \msc{x} \in ( \msc{t_1} [,] \msc{t_2} ) \leftrightarrow

  \\ 21 &&  & \vdash ( ( \msc{t_1} \le \msc{x} \wedge \msc{x} \le \msc{t_2}

     ~\le \msc{x} \wedge \msc{x} \le \msc{t_2} ) \wedge
     ~\mw{\varphi} ) )&(\mbox{\ref{eq:df-3an}})\notag
  \\ 22 &&  & \vdash ( ( \msc{x} \in \mathbb{R}^{*} \wedge ( \msc{t_1} \le
     ~\msc{x} \wedge \msc{x} \le \msc{t_2} \wedge
     ~\mw{\varphi} ) ) \leftrightarrow ( \msc{x} \in
     ~\mathbb{R}^{*} \wedge ( ( \msc{t_1} \le \msc{x}
     ~\wedge \msc{x} \le \msc{t_2} ) \wedge \mw{\varphi} )
     ~) )&(\mbox{\ref{eq:anbi2i}},21)\notag
  \\ 23 &&  & \vdash ( ( \msc{x} \in ( \msc{t_1} [,] \msc{t_2} ) \wedge
     ~\mw{\varphi} ) \leftrightarrow ( \msc{x} \in
     ~\mathbb{R}^{*} \wedge ( \msc{t_1} \le \msc{x} \wedge
     ~\msc{x} \le \msc{t_2} \wedge \mw{\varphi} ) )
     ~)&(\mbox{\ref{eq:bitr4i}},20,22)\notag
  \\ 24 &&  & \vdash ( \forall \msc{x} \in \mathbf{T} ( \msc{x} \in ( \msc{t_1}
     ~[,] \msc{t_2} ) \wedge \mw{\varphi} ) \leftrightarrow

\end{proof}
